(function() {
  var hero = document.getElementById('next-event-hero');
  if (!hero) return;
  var now = new Date();
  now.setHours(0, 0, 0, 0);
  var cards = document.querySelectorAll('.event-card[data-date]');
  var next = null;
  var nextDate = null;
  cards.forEach(function(card) {
    var d = new Date(card.dataset.date + 'T00:00:00');
    if (d >= now && (!nextDate || d < nextDate)) {
      next = card;
      nextDate = d;
    }
  });
  if (!next) return;
  var title = next.querySelector('.event-title').textContent;
  var link = next.querySelector('.event-title a');
  var time = next.querySelector('.event-time');
  var loc = next.querySelector('.event-location');
  var desc = next.querySelector('.event-desc');
  var dateEl = next.querySelector('.event-date');
  var day = next.querySelector('.event-day').textContent;
  var month = next.querySelector('.event-month').textContent;
  var weekday = next.querySelector('.event-weekday').textContent;
  var daysUntil = Math.ceil((nextDate - now) / 86400000);
  var countdown;
  if (daysUntil === 0 && time) {
    var m = time.textContent.match(/(\d{1,2}):(\d{2})/);
    if (m) {
      var eventTime = new Date();
      eventTime.setHours(parseInt(m[1]), parseInt(m[2]), 0, 0);
      var realNow = new Date();
      var diffMs = eventTime - realNow;
      if (diffMs > 0) {
        var h = Math.floor(diffMs / 3600000);
        var min = Math.floor((diffMs % 3600000) / 60000);
        countdown = h > 0 ? 'in ' + h + ' Std ' + min + ' Min' : 'in ' + min + ' Min';
      } else {
        countdown = 'Jetzt!';
      }
    } else {
      countdown = 'Heute!';
    }
  } else if (daysUntil === 0) {
    countdown = 'Heute!';
  } else if (daysUntil === 1) {
    countdown = 'Morgen';
  } else {
    countdown = 'in ' + daysUntil + ' Tagen';
  }
  hero.innerHTML =
    '<div class="hero-bar"><span>Nächster Termin</span><span class="hero-countdown">' + countdown + '</span></div>' +
    '<div class="event-card hero-card">' + next.innerHTML + '</div>';
})();

document.querySelector('.jahr-filter').addEventListener('click', function(e) {
  var btn = e.target.closest('.jahr-btn');
  if (!btn) return;
  document.querySelectorAll('.jahr-btn').forEach(function(b) { b.classList.remove('active'); });
  btn.classList.add('active');
  var year = btn.dataset.year;
  document.querySelectorAll('.termine-jahr').forEach(function(s) {
    s.style.display = (year === 'alle' || s.dataset.year === year) ? '' : 'none';
  });
});
