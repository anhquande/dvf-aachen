<script>
document.querySelector('.jahr-filter').addEventListener('click', function(e) {
  var btn = e.target.closest('.jahr-btn');
  if (!btn) return;
  document.querySelectorAll('.jahr-btn').forEach(function(b) { b.classList.remove('active'); });
  btn.classList.add('active');
  var year = btn.dataset.year;
  document.querySelectorAll('.termine-jahr').forEach(function(s) {
    s.style.display = (year === 'alle' || s.dataset.year === year) ? '' : 'none';
  });
});
</script>
